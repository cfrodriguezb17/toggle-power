cmake_minimum_required(VERSION 3.24)
project(TogglePower VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(QT_DEFAULT_MAJOR_VERSION 6)
set(QT_MAJOR_VERSION 6)
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs6)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets DBus Qml Quick Test)
find_package(KF6CoreAddons REQUIRED)
find_package(KF6I18n REQUIRED)
find_package(Plasma REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_program(MSGFMT msgfmt REQUIRED)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins")
execute_process(COMMAND bash -c "source \"$1\"; versiones_panel_comprobar \"$2\"" versions
    "${CMAKE_SOURCE_DIR}/scripts/versiones-panel-bateria.sh" "${CMAKE_SOURCE_DIR}/panel-bateria/kde-version.txt"
    RESULT_VARIABLE VERSION_CHECK)
if(NOT VERSION_CHECK EQUAL 0)
    message(FATAL_ERROR "This KDE/Qt upstream version needs a reviewed battery panel update. See README.md.")
endif()
execute_process(COMMAND pacman -Q powerdevil libplasma plasma-workspace qt6-base qt6-declarative
    OUTPUT_FILE "${CMAKE_BINARY_DIR}/kde-version.txt" COMMAND_ERROR_IS_FATAL ANY)
set(CATALOG "${CMAKE_BINARY_DIR}/share/locale/es/LC_MESSAGES/lenovo-modes.mo")
add_custom_command(OUTPUT "${CATALOG}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/share/locale/es/LC_MESSAGES"
    COMMAND ${MSGFMT} --check --check-format -o "${CATALOG}" "${CMAKE_SOURCE_DIR}/po/es.po"
    DEPENDS po/es.po)
add_custom_target(translations ALL DEPENDS "${CATALOG}")
add_library(conservation STATIC panel-bateria/conservationcontrol.cpp panel-bateria/conservationcontrol.h)
set_target_properties(conservation PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(conservation PUBLIC panel-bateria)
target_link_libraries(conservation PUBLIC Qt6::Core KF6::I18n)
add_executable(lenovo-bateria src/bateria_tray.cpp)
target_link_libraries(lenovo-bateria PRIVATE conservation Qt6::Widgets Qt6::DBus)
add_dependencies(lenovo-bateria translations)
enable_testing()
set(PLASMA_SKIP_APPLET_INSTALL TRUE)
add_subdirectory(panel-bateria)
add_executable(test-regla-bateria tests/test_regla_bateria.cpp)
target_link_libraries(test-regla-bateria PRIVATE Qt6::Core Qt6::Qml)
add_test(NAME permission-rule COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_SOURCE_DIR}/tests/check_rule.py $<TARGET_FILE:test-regla-bateria>)
add_test(NAME python COMMAND ${Python3_EXECUTABLE} -B -m unittest discover -s tests -v)
set_tests_properties(python PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" ENVIRONMENT
    "PYTHONPATH=${CMAKE_SOURCE_DIR}/src:${CMAKE_SOURCE_DIR}/scripts;LENOVO_MODES_LOCALE_DIR=${CMAKE_BINARY_DIR}/share/locale;LANGUAGE=en")
